#!/opt/homebrew/bin/python3
import os, sys, json, time
mode = os.environ['LAB_MODE']
if mode != 'clean':
    r,w = os.pipe()
    pid = os.fork()
    if pid == 0:
        os.close(r)
        if mode == 'escaped': os.setsid()
        null = os.open('/dev/null', os.O_RDWR)
        for fd in (0,1,2): os.dup2(null, fd)
        os.write(w, b'ready')
        os.close(w)
        time.sleep(1.2 if mode == 'finishing' else 3600)
        os._exit(0)
    os.close(w)
    os.read(r, 5)
    os.close(r)
    with open(os.environ['LAB_PIDFILE'], 'w') as f: f.write(str(pid))
print(json.dumps({'type':'item.completed','item':{'type':'agent_message','text':'{"verdict":"go"}'}}), flush=True)
